## Terra (ghostty, zed, topgrade, resources, kooha, opencode, ms-core-fonts…)
sudo dnf install --nogpgcheck --repofrompath 'terra,https://repos.fyralabs.com/terra$releasever' terra-release terra-gpg-keys

## RPM Fusion (free + nonfree)
sudo dnf install https://mirrors.rpmfusion.org/free/fedora/rpmfusion-free-release-$(rpm -E %fedora).noarch.rpm
sudo dnf install https://mirrors.rpmfusion.org/nonfree/fedora/rpmfusion-nonfree-release-$(rpm -E %fedora).noarch.rpm

## COPRs
sudo dnf copr enable bieszczaders/kernel-cachyos-lto        # kernel CachyOS (BORE + ThinLTO + NTSync, x86-64-v3)
sudo dnf copr enable bieszczaders/kernel-cachyos-addons     # cachyos-settings, scx-scheds, ananicy-cpp…
sudo dnf copr enable sneexy/zen-browser                     # navegador Zen
sudo dnf copr enable atim/heroic-games-launcher              # Heroic (Epic/GOG)
sudo dnf copr enable rivenirvana/morewaita-icon-theme       # MoreWaita (canal recomendado pelo upstream para Fedora)
sudo dnf copr enable chronoscrat/starship                   # prompt do shell

## Docker CE (decisão: repo do dnf, não pelo mise)
sudo dnf config-manager addrepo --from-repofile=https://download.docker.com/linux/fedora/docker-ce.repo

## Cola do boot padrão do kernel CachyOS (feito pelo script):
## /etc/dnf/libdnf5-plugins/actions.d/cachy-default.actions ->
## post_transaction:kernel*:in::/usr/bin/sh -c /usr/bin/grubby\ --set-default=/boot/$(ls\ /boot\ |\ grep\ vmlinuz.*cachy\ |\ sort\ -V\ |\ tail\ -1)
